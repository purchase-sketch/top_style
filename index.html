function applyPivot() { const tbody = document.getElementById('table-body'); const thead = document.getElementById('table-head'); if (rawData.length === 0) { tbody.innerHTML = 'Koi data nahi mila. Pehle Upload karein.'; return; } const checkedBoxes = Array.from(document.querySelectorAll('.pivot-chk:checked')); const groupCols = checkedBoxes.map(cb => cb.value); const groupLabels = checkedBoxes.map(cb => cb.getAttribute('data-label')); // Step 1: Article aur Style ke hisaab se saara data sum karna let aggregated = {}; rawData.forEach(row => { let keyObj = {}; // Grouped columns ki values (jaise Major Category, Sub Section etc.) groupCols.forEach(col => keyObj[col] = row[col] || '-'); // Article aur Style Fixed rahenge keyObj['article_name'] = row.article_name || '-'; keyObj['style_name'] = row.style_name || '-'; // Ek anokhi ID (key) banayi taaki data jud sake let keyStr = JSON.stringify(keyObj); if (!aggregated[keyStr]) { aggregated[keyStr] = { ...keyObj, sls: 0, cbs: 0 }; } // Sales aur Stock ko jodna aggregated[keyStr].sls += Number(row.net_sls_qty) || 0; aggregated[keyStr].cbs += Number(row.cbs_qty) || 0; }); // Step 2: Har Article ko alag-alag dibbe (Parent Group) me daalna let parentGroups = {}; Object.values(aggregated).forEach(row => { let parentKeyObj = {}; groupCols.forEach(col => parentKeyObj[col] = row[col]); parentKeyObj['article_name'] = row.article_name; let parentKeyStr = JSON.stringify(parentKeyObj); if (!parentGroups[parentKeyStr]) { parentGroups[parentKeyStr] = { keys: parentKeyObj, styles: [], total_sls: 0, total_cbs: 0 }; } // Us Article wale dibbe me style daal do aur Total jod do parentGroups[parentKeyStr].styles.push(row); parentGroups[parentKeyStr].total_sls += row.sls; parentGroups[parentKeyStr].total_cbs += row.cbs; }); let finalGroups = Object.values(parentGroups); // Step 3: Strict Sorting Logic (Jo aapko chahiye) // 3a. INNER SORTING: Har Article ke andar Styles ko Highest Sales to Lowest Sales karna aur sirf Top 10 rakhna finalGroups.forEach(pg => { pg.styles.sort((a, b) => b.sls - a.sls); // Style Name sorting (Highest Net SLS first) pg.styles = pg.styles.slice(0, 10); // Sirf top 10 hi dikhane hain }); // 3b. OUTER SORTING: Jis Article ka Total Net SLS sabse zyada ho, wo poora group sabse upar finalGroups.sort((a, b) => { // Pehle Total Net SLS (Descending) check karega let diff = b.total_sls - a.total_sls; if (diff !== 0) return diff; // Agar Total SLS barabar hua, toh Article Name (Alphabetical) par check karega return a.keys.article_name.localeCompare(b.keys.article_name); }); // Step 4: Table header aur Rows banana let headHtml = ''; groupLabels.forEach(lbl => headHtml += `${lbl}`); headHtml += `Article NameStyle NameNet SLS QtyCBS Qty`; thead.innerHTML = headHtml; let bodyHtml = ''; finalGroups.forEach(pg => { let rowCount = pg.styles.length; if (rowCount === 0) return; pg.styles.forEach((styleRow, index) => { bodyHtml += ``; // Rowspan Logic: Parent columns sirf pehli row me dikhenge if (index === 0) { groupCols.forEach(col => { bodyHtml += `${pg.keys[col]}`; }); bodyHtml += `${pg.keys['article_name']}`; } // Style details bodyHtml += `${styleRow.style_name}`; bodyHtml += `${styleRow.sls}`; bodyHtml += `${styleRow.cbs}`; bodyHtml += ``; }); // Subtotal Line let colspan = groupCols.length + 1; bodyHtml += ` SUBTOTAL ${pg.total_sls} ${pg.total_cbs} `; }); tbody.innerHTML = bodyHtml; }
